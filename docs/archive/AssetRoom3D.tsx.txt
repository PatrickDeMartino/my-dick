"use client";

import { useEffect, useRef } from "react";
import { SITE_ASSETS, type SiteAsset } from "../lib/assetRegistry";
import { buildProp } from "../lib/props3d";
import { loadPlayer, savePlayer } from "../lib/worldRules";
import { createAlien, updateAlien, type AlienRig } from "../../games/alien-archer/src/game/alien";
import { makeSandboxProp } from "../../games/alien-archer/src/game/sandbox";

export type AssetMode = "display" | "ragdoll" | "function";
type Spawned = { id:number; root:import("three").Group; velocity:import("three").Vector3; spin:import("three").Vector3; asset:SiteAsset; radius:number; baseScale:number; mode:AssetMode; alien?:AlienRig; grounded:boolean; heldBy?:Spawned };
type Status = { id:number; name:string; mode:AssetMode; scale:number } | null;

export default function AssetRoom3D({onCount,onActive,onSelected}:{onCount:(n:number)=>void;onActive:(s:string)=>void;onSelected:(s:Status)=>void}){
 const mountRef=useRef<HTMLDivElement>(null);
 useEffect(()=>{const mount=mountRef.current;if(!mount)return;let disposed=false,cleanup=()=>{};
 void Promise.all([import("three"),import("three/examples/jsm/controls/OrbitControls.js")]).then(([THREE,{OrbitControls}])=>{if(disposed)return;
  const scene=new THREE.Scene();scene.background=new THREE.Color(0x04080e);scene.fog=new THREE.Fog(0x04080e,18,44);
  const camera=new THREE.PerspectiveCamera(52,1,.1,90);camera.position.set(9,7.5,12);
  const renderer=new THREE.WebGLRenderer({antialias:true,powerPreference:"high-performance"});renderer.setPixelRatio(Math.min(devicePixelRatio,1.6));renderer.shadowMap.enabled=true;renderer.outputColorSpace=THREE.SRGBColorSpace;mount.appendChild(renderer.domElement);
  const controls=new OrbitControls(camera,renderer.domElement);controls.target.set(0,1,0);controls.enableDamping=true;controls.minDistance=4;controls.maxDistance=30;controls.maxPolarAngle=Math.PI*.49;
  scene.add(new THREE.HemisphereLight(0xa9eaff,0x190d27,1.7));const key=new THREE.DirectionalLight(0xffffff,3);key.position.set(7,13,6);key.castShadow=true;scene.add(key);const glow=new THREE.PointLight(0x5dffd7,20,20,2);glow.position.set(-5,4,-4);scene.add(glow);
  const deck=new THREE.Mesh(new THREE.CylinderGeometry(8,8.6,.65,8),new THREE.MeshStandardMaterial({color:0x12292d,roughness:.42,metalness:.65}));deck.receiveShadow=true;scene.add(deck);const grid=new THREE.GridHelper(14,14,0x68ffe1,0x24484e);grid.position.y=.34;scene.add(grid);
  for(let i=0;i<10;i++){const a=i/10*Math.PI*2,c=new THREE.Mesh(new THREE.ConeGeometry(.18,.9,6),new THREE.MeshPhysicalMaterial({color:new THREE.Color().setHSL(i/10,.85,.58),emissive:new THREE.Color().setHSL(i/10,.7,.2),emissiveIntensity:.7,transparent:true,opacity:.82,roughness:.1}));c.position.set(Math.cos(a)*7.3,.76,Math.sin(a)*7.3);scene.add(c)}
  const spawned:Spawned[]=[];let nextId=1,selected:Spawned|null=null,controlled:Spawned|null=null,held:Spawned|null=null,ammo="pepsi";const keys=new Set<string>();const ray=new THREE.Raycaster(),pointer=new THREE.Vector2();
  const report=(entity:Spawned|null)=>{selected=entity;onSelected(entity?{id:entity.id,name:entity.asset.label,mode:entity.mode,scale:entity.root.scale.x/entity.baseScale}:null)};
  const mode=(entity:Spawned,value:AssetMode)=>{entity.mode=value;entity.velocity.set(0,0,0);entity.grounded=false;if(value==="display"){entity.root.position.y=Math.max(entity.root.position.y,2);if(entity===controlled){controlled=null;onActive("NONE")}}else if(value==="function"){if(entity.asset.category==="CHARACTERS"&&entity.alien){controlled=entity;savePlayer(entity.asset.character!);onActive(entity.asset.label);controls.target.copy(entity.root.position)}else if(entity.asset.category==="VEHICLES"){controlled=entity;onActive(entity.asset.label)}}report(entity)};
  const spawn=(asset:SiteAsset,initial:AssetMode="display",position?:import("three").Vector3)=>{let root:import("three").Group,alien:AlienRig|undefined;if(asset.builder==="alien"&&asset.character){alien=createAlien(asset.character);root=alien.root}else if(asset.builder==="shared-prop"&&asset.prop)root=buildProp(THREE,asset.prop);else root=makeSandboxProp(asset.sandbox??"vehicle");const base=asset.scale??1;root.scale.setScalar(base);root.position.copy(position??new THREE.Vector3((Math.random()-.5)*7,2.2+Math.random(),(Math.random()-.5)*7));root.traverse(n=>{if(n instanceof THREE.Mesh){n.castShadow=true;n.receiveShadow=true;n.userData.entityRoot=root}});scene.add(root);const entity:Spawned={id:nextId++,root,velocity:new THREE.Vector3(),spin:new THREE.Vector3(.18,.55,.12),asset,radius:asset.builder==="alien"?.55:.4,baseScale:base,mode:initial,alien,grounded:false};spawned.push(entity);onCount(spawned.length);report(entity);mode(entity,initial);return entity};
  const clear=()=>{spawned.forEach(e=>scene.remove(e.root));spawned.length=0;selected=controlled=held=null;onCount(0);onActive("NONE");onSelected(null)};
  const shoot=()=>{if(!controlled)return;const asset=SITE_ASSETS.find(a=>a.id===ammo)||SITE_ASSETS.find(a=>a.id==="pepsi")!;const d=new THREE.Vector3(0,0,-1).applyQuaternion(controlled.root.quaternion);const shot=spawn(asset,"ragdoll",controlled.root.position.clone().add(new THREE.Vector3(0,1,0)).addScaledVector(d,.9));shot.velocity.copy(d.multiplyScalar(15));shot.velocity.y=1.5};
  const pickup=()=>{if(!controlled)return;if(held){held.heldBy=undefined;held.mode="ragdoll";held.velocity.copy(controlled.velocity);held=null;return}let near:Spawned|null=null,best=2.3;for(const e of spawned){if(e===controlled||e.asset.category==="CHARACTERS")continue;const d=e.root.position.distanceTo(controlled.root.position);if(d<best){best=d;near=e}}if(near){held=near;near.heldBy=controlled;near.mode="function";near.velocity.set(0,0,0)}};
  const onSpawn=(ev:Event)=>{const d=(ev as CustomEvent<{id:string;mode?:AssetMode}>).detail,a=SITE_ASSETS.find(x=>x.id===d.id);if(a)spawn(a,d.mode??"display")};const onMode=(ev:Event)=>{if(selected)mode(selected,(ev as CustomEvent<{mode:AssetMode}>).detail.mode)};const onScale=(ev:Event)=>{if(selected){const scale=(ev as CustomEvent<{scale:number}>).detail.scale;selected.root.scale.setScalar(selected.baseScale*scale);report(selected)}};const onAmmo=(ev:Event)=>{ammo=(ev as CustomEvent<{id:string}>).detail.id};
  const press=(code:string)=>{if(code==="Space"&&controlled?.grounded){controlled.velocity.y=6.5;controlled.grounded=false}if(code==="KeyF")shoot();if(code==="KeyE")pickup()};const onKey=(e:KeyboardEvent)=>{if(["KeyW","KeyA","KeyS","KeyD","Space","ShiftLeft","KeyF","KeyE"].includes(e.code))e.preventDefault();keys.add(e.code);if(!e.repeat)press(e.code)};const onUp=(e:KeyboardEvent)=>keys.delete(e.code);const onVirtual=(ev:Event)=>{const d=(ev as CustomEvent<{code:string;down:boolean}>).detail;d.down?(keys.add(d.code),press(d.code)):keys.delete(d.code)};
  const onPointer=(e:PointerEvent)=>{if((e.target as HTMLElement).closest("button,a,aside"))return;const rect=renderer.domElement.getBoundingClientRect();pointer.set((e.clientX-rect.left)/rect.width*2-1,-(e.clientY-rect.top)/rect.height*2+1);ray.setFromCamera(pointer,camera);const hits=ray.intersectObjects(spawned.map(x=>x.root),true);if(hits[0]){const root=hits[0].object.userData.entityRoot;report(spawned.find(x=>x.root===root)??null)}};
  window.addEventListener("asset-room:spawn",onSpawn);window.addEventListener("asset-room:clear",clear);window.addEventListener("asset-room:set-mode",onMode);window.addEventListener("asset-room:set-scale",onScale);window.addEventListener("asset-room:set-ammo",onAmmo);window.addEventListener("asset-room:key",onVirtual);window.addEventListener("keydown",onKey);window.addEventListener("keyup",onUp);renderer.domElement.addEventListener("pointerdown",onPointer);
  const resize=()=>{const w=mount.clientWidth,h=mount.clientHeight;camera.aspect=w/Math.max(1,h);camera.updateProjectionMatrix();renderer.setSize(w,h,false)};const observer=new ResizeObserver(resize);observer.observe(mount);resize();const clock=new THREE.Clock();
  renderer.setAnimationLoop(()=>{const dt=Math.min(clock.getDelta(),.035),time=clock.elapsedTime;for(const e of spawned){if(e.heldBy){e.root.position.copy(e.heldBy.root.position).add(new THREE.Vector3(.62,1.05,-.42).applyQuaternion(e.heldBy.root.quaternion));e.root.quaternion.copy(e.heldBy.root.quaternion);continue}if(e.mode==="display"){e.root.rotation.y+=dt*.62;e.root.position.y+=Math.sin(time*1.7+e.id)*dt*.045;continue}const active=e===controlled&&e.mode==="function";let speed=0;if(active){const x=(keys.has("KeyD")?1:0)-(keys.has("KeyA")?1:0),z=(keys.has("KeyS")?1:0)-(keys.has("KeyW")?1:0),boost=keys.has("ShiftLeft")?8:4.7;e.velocity.x+=(x*boost-e.velocity.x)*Math.min(1,dt*9);e.velocity.z+=(z*boost-e.velocity.z)*Math.min(1,dt*9);speed=Math.hypot(e.velocity.x,e.velocity.z);if(speed>.1)e.root.rotation.y=Math.atan2(e.velocity.x,e.velocity.z)+Math.PI;if(e.asset.category==="VEHICLES"&&keys.has("Space"))e.velocity.y+=9*dt}else{e.root.rotation.x+=e.spin.x*dt;e.root.rotation.y+=e.spin.y*dt;e.root.rotation.z+=e.spin.z*dt}e.velocity.y-=e.asset.category==="VEHICLES"&&active?3.2*dt:12*dt;e.root.position.addScaledVector(e.velocity,dt);const floor=.34+e.radius;if(e.root.position.y<floor){e.root.position.y=floor;e.velocity.y=Math.abs(e.velocity.y)>1?.18:0;e.grounded=true;e.velocity.x*=active?.84:.94;e.velocity.z*=active?.84:.94}for(const axis of ["x","z"] as const)if(Math.abs(e.root.position[axis])>7.2){e.root.position[axis]=Math.sign(e.root.position[axis])*7.2;e.velocity[axis]*=-.45}if(e.alien)updateAlien(e.alien,{dt,speed,grounded:e.grounded,vy:e.velocity.y,justLanded:false,justShot:false,attract:false,time,weapon:"revolver",firing:false,aimPitch:0,reload:0,revLoaded:6,jetting:false})}controls.update();renderer.render(scene,camera)});
  const saved=loadPlayer();const initial=SITE_ASSETS.find(a=>a.character===saved)??SITE_ASSETS[0];spawn(initial,"function",new THREE.Vector3(0,2.4,0));
  cleanup=()=>{renderer.setAnimationLoop(null);observer.disconnect();controls.dispose();window.removeEventListener("asset-room:spawn",onSpawn);window.removeEventListener("asset-room:clear",clear);window.removeEventListener("asset-room:set-mode",onMode);window.removeEventListener("asset-room:set-scale",onScale);window.removeEventListener("asset-room:set-ammo",onAmmo);window.removeEventListener("asset-room:key",onVirtual);window.removeEventListener("keydown",onKey);window.removeEventListener("keyup",onUp);renderer.dispose();renderer.domElement.remove()};
 });return()=>{disposed=true;cleanup()};},[onActive,onCount,onSelected]);
 return <div ref={mountRef} className="asset-room__viewport" aria-label="3D asset physics test platform"/>;
}
